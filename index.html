const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());
app.use(express.static('public'));

// 1. MONGODB ATLAS ULANISHI
mongoose.connect(process.env.MONGO_URI)
  .then(() => console.log('MongoDB Atlas muvaffaqiyatli ulandi'))
  .catch(err => console.error('MongoDB xato:', err));

const LeadSchema = new mongoose.Schema({
  igId: { type: String, unique: true },
  name: String,
  username: String,
  account: { type: String, default: '@toylarga.taklfinoma' },
  status: { type: String, default: 'new' },
  priority: { type: String, default: 'medium' },
  assignee: { type: String, default: 'Asosiy Menejer' },
  phone: String,
  service: String,
  amount: { type: Number, default: 0 },
  eventDate: String,
  paymentStatus: { type: String, default: 'unpaid' },
  followUp: String,
  adCampaign: String,
  notes: String,
  unread: { type: Boolean, default: true },
  messages: [{ sender: String, text: String, time: String, internal: Boolean }]
}, { timestamps: true });

const Lead = mongoose.model('Lead', LeadSchema);

// 2. META INSTAGRAM WEBHOOK TEKSHIRUVI (GET)
app.get('/webhook/instagram', (req, res) => {
  if (req.query['hub.verify_token'] === process.env.META_VERIFY_TOKEN) {
    return res.status(200).send(req.query['hub.challenge']);
  }
  res.sendStatus(403);
});

// 3. INSTAGRAM DIRECT'DAN YANGI XABAR KELGANDA (POST)
app.post('/webhook/instagram', async (req, res) => {
  res.sendStatus(200);
  const entry = req.body.entry?.[0];
  const recipientPageId = entry?.id;
  const event = entry?.messaging?.[0];
  if (!event || !event.message || event.message.is_echo) return;

  const senderId = event.sender.id;
  const userText = event.message.text || '[Media fayl]';
  const timeNow = new Date().toLocaleTimeString('uz-UZ', { hour: '2-digit', minute: '2-digit' });

  // Qaysi Instagram sahifaga yozilganini aniqlash
  const accountName = recipientPageId === process.env.META_PAGE_ID_2
    ? '@toyga.taklifnoma'
    : '@toylarga.taklfinoma';

  let lead = await Lead.findOne({ igId: senderId });
  if (!lead) {
    lead = await Lead.create({
      igId: senderId,
      name: 'Instagram Mijoz #' + senderId.slice(-4),
      username: 'ig_' + senderId.slice(-5),
      account: accountName,
      messages: []
    });
  }

  lead.unread = true;
  lead.messages.push({ sender: 'user', text: userText, time: timeNow });
  await lead.save();

  // Telegram botga xabar yuborish
  if (process.env.TG_BOT_TOKEN && process.env.TG_CHAT_ID) {
    fetch(`https://api.telegram.org/bot${process.env.TG_BOT_TOKEN}/sendMessage`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        chat_id: process.env.TG_CHAT_ID,
        text: `📩 Yangi xabar (${accountName})!\nMijoz: ${lead.name}\nXabar: ${userText}`
      })
    }).catch(() => {});
  }

  // Gemini AI orqali avtomatik javob berish
  if (process.env.GEMINI_API_KEY) {
    const history = lead.messages.map(m => `${m.sender === 'user' ? 'Mijoz' : 'Menejer'}: ${m.text}`).join('\n');
    const aiRes = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${process.env.GEMINI_API_KEY}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: `Sen tirik o'zbek menejerisan. Qisqa va samimiy javob yoz.\nSuhbat:\n${history}` }] }]
      })
    });
    const aiData = await aiRes.json();
    const replyText = aiData.candidates?.[0]?.content?.parts?.[0]?.text;
    if (replyText) {
      await sendInstagramDM(senderId, replyText, accountName);
      lead.messages.push({ sender: 'ai', text: replyText, time: timeNow });
      await lead.save();
    }
  }
});

// INSTAGRAM GRAPH API ORQALI JAVOB YUBORISH (IKKALA SAHIFA UCHUN)
async function sendInstagramDM(recipientId, text, accountName) {
  const token = accountName === '@toyga.taklifnoma'
    ? (process.env.META_TOKEN_2 || process.env.META_PAGE_TOKEN)
    : (process.env.META_TOKEN_1 || process.env.META_PAGE_TOKEN);

  if (!token) return;
  await fetch(`https://graph.facebook.com/v21.0/me/messages?access_token=${token}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ recipient: { id: recipientId }, message: { text } })
  });
}

// CRM FRONTEND API ROUTES
app.get('/api/leads', async (req, res) => res.json(await Lead.find().sort({ updatedAt: -1 })));
app.post('/api/leads', async (req, res) => res.json(await Lead.create(req.body)));
app.put('/api/leads/:id', async (req, res) => res.json(await Lead.findByIdAndUpdate(req.params.id, req.body, { new: true })));
app.delete('/api/leads/:id', async (req, res) => res.json(await Lead.findByIdAndDelete(req.params.id)));
app.post('/api/send-message', async (req, res) => {
  const { leadId, igId, text, internal } = req.body;
  const timeNow = new Date().toLocaleTimeString('uz-UZ', { hour: '2-digit', minute: '2-digit' });
  const existing = await Lead.findById(leadId);
  if (!internal && igId) await sendInstagramDM(igId, text, existing?.account);
  const lead = await Lead.findByIdAndUpdate(leadId, { $push: { messages: { sender: 'admin', text, time: timeNow, internal } } }, { new: true });
  res.json(lead);
});

app.listen(process.env.PORT || 3000, () => console.log('InstaCRM Server 3000-portda ishga tushdi'));
